@app.route("/api/rules", methods=["GET"])
def get_rules():
    try:
        with open('rules.json', "r", encoding="utf-8") as f:
            data = json.load(f)
        return jsonify(data)
    except Exception as e:
        return jsonify({"error": str(e)}), 500

@app.route("/api/rules", methods=["POST"])
def update_rules():
    try:
        data = request.get_json()
        new_rules = data.get("rules", [])
        
        with open('rules.json', "w", encoding="utf-8") as f:
            json.dump({"rules": new_rules}, f, ensure_ascii=False, indent=4)
        
        return jsonify({"message": "✅ Cập nhật thành công!", "status": "success"})
    except Exception as e:
        return jsonify({"message": f"❌ Lỗi: {str(e)}", "status": "error"}), 500

useEffect(() => {
  if (authorized) {
    fetch("http://127.0.0.1:5000/api/rules")
      .then((res) => res.json())
      .then((data) => {
        setRules(data.rules || []);
        setHasChanges(false);
      })
      .catch((err) => {
        console.error("Lỗi tải rules:", err);
        alert("❌ Không thể tải dữ liệu luật từ server!");
      });
  }
}, [authorized]);

const saveChanges = async () => {
  try {
    const response = await fetch("http://127.0.0.1:5000/api/rules", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ rules }),
    });
    
    const data = await response.json();
    
    if (response.ok) {
      alert("✅ " + data.message);
      setHasChanges(false);
    } else {
      alert("❌ " + (data.message || "Có lỗi xảy ra!"));
    }
  } catch (err) {
    alert("❌ Lỗi kết nối đến server: " + err.message);
  }
};
